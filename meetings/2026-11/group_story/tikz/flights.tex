% Two flights: a solo flight (S), and a flight that is solo (A), joins a group (G) and is solo again (A); the populations are named on the drawing.
\begin{tikzpicture}[x=0.9cm,y=0.55cm,line width=1.6pt]
  % top: solo flight
  \draw[blueI,domain=0:10,samples=100,smooth,variable=\x] plot (\x,{2.6+0.20*sin(5*\x r)+0.12*sin(12.5*\x r)});
  \node[blueI,anchor=east,font=\Large\bfseries] at (-0.2,2.6) {S};
  \node[anchor=south west,font=\small,text=ink2] at (0,2.95) {a flight never in a group};
  % bottom: A, G (with companions), A
  \draw[aqua,domain=0:3.3,samples=40,smooth,variable=\x] plot (\x,{0.2+0.20*sin(5.5*\x r)+0.12*sin(13*\x r)});
  \foreach \d/\p in {0.55/4.6,-0.5/6.1,0.05/3.7}{
    \draw[muted,line width=0.9pt,domain=3.3:6.7,samples=40,smooth,variable=\x]
      plot (\x,{0.2+\d+0.20*sin(\p*\x r)+0.12*sin((2.3*\p+1.7)*\x r)});}
  \draw[orangeJ,domain=3.3:6.7,samples=40,smooth,variable=\x] plot (\x,{0.2+0.20*sin(5.5*\x r)+0.12*sin(13*\x r)});
  \draw[aqua,domain=6.7:10,samples=40,smooth,variable=\x] plot (\x,{0.2+0.20*sin(5.5*\x r)+0.12*sin(13*\x r)});
  \node[anchor=south west,font=\small,text=ink2] at (0,1.0) {a flight with group time};
  % the three stretches of the bottom flight, named under it
  \foreach \a/\b/\c/\t in {0.05/3.25/aqua/{A: solo time},3.35/6.65/orangeJ/{G: group time},6.75/9.95/aqua/{A: solo time}}{
    \draw[decorate,decoration={brace,amplitude=4pt,mirror},line width=0.8pt,\c] (\a,-0.55) -- (\b,-0.55)
      node[midway,below=3pt,font=\small\bfseries,text=\c] {\t};}
  \node[font=\footnotesize,text=muted,anchor=west] at (6.75,1.1) {other pilots};
\end{tikzpicture}
