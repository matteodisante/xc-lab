\begin{frame}[label=t-pieces]{Methods · pieces, TA-MSD and $\alpha(\tau)$}
\back{m-pool}
\small

{\footnotesize
\[ \overline{\delta^2_p}(\tau)=\frac{1}{M-k}\sum_{j=0}^{M-k-1}|x_{j+k}-x_j|^2,\qquad \MSD(\tau)=\frac1N\sum_p\overline{\delta^2_p}(\tau) \]
\[ \alpha(\tau)=\frac12\,\frac{\sum_{k\in W}(u_k-\bar u)(y_k-\bar y)}{\sum_{k\in W}(u_k-\bar u)^2},\qquad u_k=\log\tau_k,\ y_k=\log\MSD(\tau_k) \]}

  \item A piece lasts at least $\tau_\mathrm{max}/\Support$: at least \SupportOriginsPercent\% of it gives origins at $\tau_\mathrm{max}$.
\end{itemize}
\takeaway{$\alpha(\tau)$ is half the log-slope of the MSD in a window around $\tau$, on pieces that all reach the lags of the window.}
\end{frame}

\begin{frame}[label=t-simpson]{Where? · Simpson's paradox: the algebra}
\back{m-ground}
Populations $P$, with class shares $w_c^P$ ($\sum_c w_c^P=1$) and a value $x_c^P$ in each class $c$:
\[ X^P=\sum_c w_c^P\,x_c^P \]
\[ X^\A-X^\Sl=\underbrace{\sum_c w_c^\A\,\bigl(x_c^\A-x_c^\Sl\bigr)}_{\text{within the classes}}
 +\underbrace{\sum_c \bigl(w_c^\A-w_c^\Sl\bigr)\,x_c^\Sl}_{\text{composition}} \]
If $x_c^\A>x_c^\Sl$ in every class, the first term is positive. If $\A$ weighs more on the classes where $x$ is low, the second is negative and can win: the overall difference flips sign.

\medskip
Classic example: a hospital better at easy and at hard operations has the lower overall survival if it operates many more hard cases.
\takeaway{Compare at fixed class: the sign on the mix of classes reads where the populations are, not how they differ.}
\end{frame}

\begin{frame}[label=t-readings]{Direction · the two readings and what each predicts}
\back{m-direction}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}\small
\textbf{group $\to$ straight}: group time finds, selects or holds straighter paths. The persistence rises at the entry and falls at the exit.
\column{.48\textwidth}\small
\textbf{straight $\to$ group}: on a straight leg the paths converge, so pilots meet and stay together. The window before the group is already as persistent as the start; the window after is less.
\end{columns}
\[ \text{before}\to\text{start: }\mv{\WhenAllLongStartMinusPre}\ \mv{\WhenAllLongStartMinusPreBand},\qquad \text{end}\to\text{after: }\mv{\WhenAllLongPostMinusEnd}\ \mv{\WhenAllLongPostMinusEndBand} \]
\begin{itemize}\setlength{\itemsep}{0.3em}\small
  \item The rise at the entry is resolved on every flight and in the high mountains; both steps survive the control events.
  \item The exit does not separate the two readings.
  \item To do: the timing at a shorter scale; solo pilots on the same leg and hour; groups that leave the same takeoff together.
\end{itemize}
\takeaway{The entry points to group $\to$ straight, for legs of a window (\WhenWindow~s); a shorter leg would not show.}
\end{frame}

\begin{frame}[label=t-found]{What we found · in full}
\back{m-found}
\begin{enumerate}\setlength{\itemsep}{0.45em}\normalsize
  \item Group time is more persistent from a few hundred seconds on.
  \item At long lags the gap is $\G-\A$, within the same flights.
  \item The absolute effect grows with the relief.
  \item The opening acts on $\A-\Sl$; within a flight the effect survives it.
  \item $\A-\Sl$ comes neither from the pilots nor from carry-over.
  \item Every flight: persistence rises at the entry, falls at the exit; per class the fall is clearer.
  \item More climbs, found more reliably, better glides; not the wind.
  \item Mostly in the shape of the route; the timing adds little.
  \item The entry points to group $\to$ straight: the persistence rises after joining, not before.
\end{enumerate}
\takeaway{Group time is straighter than solo time, on every ground, at a scale set by the ground.}
\end{frame}
