\begin{frame}[label=t-pieces]{Methods · pieces, TA-MSD and $\alpha(\tau)$}
\back{m-pool}
\small
Piece: the part of a segment in one population; only pieces of at least \PieceMinS~s ($=1\,000/\Support$) are kept.
{\footnotesize
\[ \overline{\delta^2_p}(\tau)=\frac{1}{M-k}\sum_{j=0}^{M-k-1}|x_{j+k}-x_j|^2,\qquad \MSD(\tau)=\frac1N\sum_p\overline{\delta^2_p}(\tau) \]
\[ \alpha(\tau)=\frac12\,\frac{\sum_{k\in W}(u_k-\bar u)(y_k-\bar y)}{\sum_{k\in W}(u_k-\bar u)^2},\qquad u_k=\log\tau_k,\ y_k=\log\MSD(\tau_k) \]}
\begin{itemize}\setlength{\itemsep}{0.3em}\footnotesize
  \item Positions on a grid of \GridS~s; one weight per piece; \LagCount{} lags, geometric, from \LagMin{} to \LagMax~s.
  \item Window $W(\tau)$: the lags within \HalfDecade{} decades of $\tau$.
  \item A piece lasts at least $\tau_\mathrm{max}/\Support$: at least \SupportOriginsPercent\% of it gives origins at $\tau_\mathrm{max}$.
\end{itemize}
\takeaway{$\alpha(\tau)$ is half the log-slope of the MSD in a window around $\tau$, on pieces that all reach the lags of the window.}
\end{frame}

\begin{frame}[label=t-bands]{Methods · the bands}
\back{m-persistent}
\small
Pieces are not independent: pilots share skill and wing, blocks (a group episode, or one site and day) share the air.
\[ V=\max(V_p+V_b-V_c,\;V_p,\;V_b),\qquad \text{band}=\text{estimate}\pm\BandZ\sqrt V\ \ (90\%) \]
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Three bootstraps of \NBoot{} draws resample the pilots, the blocks and the cells (pilot $\times$ block); $V_p$, $V_b$, $V_c$ are the variances over them (Cameron, Gelbach, Miller).

  \item On synthetic data with pilot and block effects, the nominal 90\% band covers in \CovNewMin--\CovNewMax\% of the datasets.
\end{itemize}
\takeaway{Every band of the talk is a 90\% band that counts pilots and blocks, not pieces.}
\end{frame}

\begin{frame}[label=t-simpson]{Where? · Simpson's paradox: the algebra}
\back{m-ground}
Populations $P$, with class shares $w_c^P$ ($\sum_c w_c^P=1$) and a value $x_c^P$ in each class $c$:
\[ X^P=\sum_c w_c^P\,x_c^P \]
\[ X^\A-X^\Sl=\underbrace{\sum_c w_c^\A\,\bigl(x_c^\A-x_c^\Sl\bigr)}_{\text{within the classes}}
 +\underbrace{\sum_c \bigl(w_c^\A-w_c^\Sl\bigr)\,x_c^\Sl}_{\text{composition}} \]
If $x_c^\A>x_c^\Sl$ in every class, the first term is positive. If $\A$ weighs more on the classes where $x$ is low, the second is negative and can win: the overall difference flips sign.

\medskip
Classic example: a hospital better at easy and at hard operations has the lower overall survival if it operates many more hard cases.
\takeaway{Compare at fixed class: the sign on the mix of classes reads where the populations are, not how they differ.}
\end{frame}

\begin{frame}[label=t-memory]{Where? · the memory of the route: method}
\back{m-memory}
\begin{itemize}\small
  \item Route of each piece: the path on a 1-s grid, each climb run replaced by its chord, the position moving along the chord in real time.
  \item Heading $\theta(t)$ of the 10-s displacement, undefined below 20~m (the slow chords of the climbs drop out).
  \item $C(\tau)=\langle\cos(\theta(t+\tau)-\theta(t))\rangle$ over origins every 10~s, pooled over the pieces of a fixed cohort (at least 4\,500~s, so every lag up to \LegTmax~s is on the same pieces).
  \item The memory time is $\tau_c$ at $C=1/e$; the table also gives $\tau_{1/2}$ at $C=1/2$, and the same along the route in metres ($s_c$).
  \item Bands: 90\,\%, bootstrap over flights, independent for each class and population: $\G-\A$ is read from non-overlapping bands.
  \item The main slide draws the chained curve (\hyperlink{t-chainedC}{$C^*$ frame}); the table and the cohort above are the fixed cohort.
\end{itemize}
\[ \MSD(\tau)=2v^2\!\int_0^\tau(\tau-s)\,C(s)\,ds\quad\text{(constant speed $v$): } \alpha=1 \text{ while } C\approx1 \]
\takeaway{How long the route holds its direction, in time and in space: the relief shortens both. In the plains $C$ barely decays and $\G$, $\A$ keep it alike: there the gap is not in the direction.}
\end{frame}

\chap{0}
\begin{frame}[label=t-chainedC]{Where? · the chained correlation of the direction}
\backs{m-memory}{main frame}{t-memory}{method}
\footnotesize
$s_p(\tau)=\sum_t\cos\bigl(\theta(t+\tau)-\theta(t)\bigr)$ over the origins of piece $p$, $n_p(\tau)$ their number; $P(\tau)$: the pieces that reach $\tau$.
\[ C_S(\tau)=\frac{\sum_{p\in S}s_p(\tau)}{\sum_{p\in S}n_p(\tau)},\qquad
   C^*(\tau_1)=C_{P(\tau_1)}(\tau_1),\qquad C^*(\tau_{k+1})=C^*(\tau_k)\,\frac{C_{P(\tau_{k+1})}(\tau_{k+1})}{C_{P(\tau_{k+1})}(\tau_k)} \]
\begin{itemize}\setlength{\itemsep}{0.25em}
  \item \emph{Why:} the pool $C_{P(\tau)}(\tau)$ loses the short pieces as $\tau$ grows and rises with who is left, as the MSD; a fixed cohort stops at its own length.
  \item \emph{Each factor} compares two lags on one set, $P(\tau_{k+1})\subseteq P(\tau_k)$: no composition term. If every piece decays as $f(\tau)$, the product telescopes to $f(\tau)$ whichever pieces leave (product-limit, as Kaplan--Meier).
  \item \emph{Limits:} it stops where the denominator nears zero or fewer than 30 pieces reach the lag. Bands from the flights, $\G$ and $\A$ of a flight drawn together: $C^*_\G-C^*_\A$ has its own band.
\end{itemize}
\takeaway{One correlation curve per population up to the longest lags, without the bias of who is left: the analogue of $\alpha^*$ for the direction.}
\end{frame}

\begin{frame}[label=t-readings]{Cause? · the two readings and what each predicts}
\back{m-direction}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}\small
\textbf{group $\to$ straight}: group time finds, selects or holds straighter paths. The persistence rises at the entry and falls at the exit.
\column{.48\textwidth}\small
\textbf{straight $\to$ group}: on a straight leg the paths converge, so pilots meet and stay together. The window before the group is already as persistent as the start; the window after is less.
\end{columns}
\[ \text{before}\to\text{start: }\mv{\WhenAllLongStartMinusPre}\ \mv{\WhenAllLongStartMinusPreBand},\qquad \text{end}\to\text{after: }\mv{\WhenAllLongPostMinusEnd}\ \mv{\WhenAllLongPostMinusEndBand} \]

\begin{frame}[label=t-hillmethod]{Methods · tails: the Hill exponent and its threshold}
\back{m-tails}
\small
\[ P(X>x)\sim x^{-\mu},\qquad \hat\mu=\frac{n}{\sum_{i:\,x_i\ge x_\mathrm{min}}\ln(x_i/x_\mathrm{min})} \]
\begin{itemize}\setlength{\itemsep}{0.5em}
  \item Durations of glides, of searches that end in a climb and of climbs.
  \item Threshold of a curve: the $x_\mathrm{min}$ that minimises the Kolmogorov--Smirnov distance to the fitted power law (Clauset, Shalizi, Newman).
  \item We read $\mu$ of $\Sl$, $\A$, $\G$ above one threshold per phase, the largest of their three, re-chosen in every draw: the three exponents describe the same durations and the identity $\G-\Sl=(\G-\A)+(\A-\Sl)$ stays exact.
\end{itemize}
\takeaway{One threshold for the three populations: the gap of $\mu$ does not mix the range with the population.}
\end{frame}

\secframe{s-summary}{Summary}{what we found in full, what the data do not say}

\chap{0}
\begin{frame}[label=t-found]{What we found · in full}
\back{m-found}
\begin{enumerate}\setlength{\itemsep}{0.45em}\normalsize
  \item Group time is more persistent from a few hundred seconds on.
  \item At long lags the gap is $\G-\A$, within the same flights.
  \item The absolute effect grows with the relief.
  \item The opening acts on $\A-\Sl$; within a flight the effect survives it.
  \item $\A-\Sl$ comes neither from the pilots nor from carry-over.
  \item Every flight: persistence rises at the entry, falls at the exit; per class the fall is clearer.
  \item More climbs, found more reliably, better glides; not the wind.
  \item Mostly in the shape of the route; the timing adds little.
  \item The entry points to group $\to$ straight: the persistence rises after joining, not before.
\end{enumerate}
\takeaway{Group time is straighter than solo time, on every ground, at a scale set by the ground.}
\end{frame}
